function abrirMovimientoCaja(tipo) {
  movimientoTipoActual = tipo;
  document.getElementById("mov-titulo").textContent = tipo === "ingreso" ? "➕ Nuevo ingreso" : "➖ Nuevo egreso";
  document.getElementById("mov-concepto").value = "";
  document.getElementById("mov-monto").value = "";
  document.getElementById("mov-categoria").value = "";
  document.getElementById("modal-movimiento").classList.add("active");
  setTimeout(() => document.getElementById("mov-concepto").focus(), 200);
}
function guardarMovimientoCaja() {
  const concepto = document.getElementById("mov-concepto").value.trim();
  const monto = parseFloat(document.getElementById("mov-monto").value);
  const categoria = document.getElementById("mov-categoria").value.trim() || "General";
  if (!concepto || isNaN(monto) || monto <= 0) return alert("Completa concepto y monto válido");
  agregarMovimiento(movimientoTipoActual, monto, concepto, categoria);
  guardar(); renderTodo();
  cerrarModal("modal-movimiento");
  mostrarEstadoSync("ok", "Movimiento registrado");
}
function eliminarMovimiento(id) {
  if (!confirm("¿Eliminar este movimiento de caja?")) return;
  datos.caja.movimientos = datos.caja.movimientos.filter(m => m.id !== id);
  guardar(); renderTodo();
}
function editarAperturaCaja() {
  const actual = datos.caja.apertura || 0;
  const nuevo = prompt("Apertura inicial de caja ($):", actual);
  if (nuevo === null) return;
  const val = parseFloat(nuevo);
  if (isNaN(val)) return alert("Valor inválido");
  datos.caja.apertura = val;
  guardar(); renderTodo();
  alert("✅ Apertura actualizada");
}

function seleccionarMedioPago(el) {
  const padre = el.parentElement;
  padre.querySelectorAll(".medio-pago").forEach(m => m.classList.remove("selected"));
  el.classList.add("selected");
}
function obtenerMedioPagoSeleccionado(contenedorId) {
  const el = document.getElementById(contenedorId);
  if (!el) return "efectivo";
  const sel = el.querySelector(".medio-pago.selected");
  return sel ? sel.dataset.medio : "efectivo";
}

function registrarVenta() {
  const producto = document.getElementById("v-producto").value.trim();
  const categoria = document.getElementById("v-categoria").value.trim() || "General";
  const costo = parseFloat(document.getElementById("v-costo").value) || 0;
  const precio = parseFloat(document.getElementById("v-precio").value);
  const tipo = document.getElementById("v-tipo").value;
  const cliente = document.getElementById("v-cliente").value.trim() || (tipo==="contado"?"Contado":"Cliente");
  const quincena = tipo === "quincena" ? document.getElementById("v-quincena").value : "";
  const codigo = document.getElementById("v-codigo").value.trim();
  const metodoPago = obtenerMedioPagoSeleccionado("medios-pago-venta");

  if (!producto || isNaN(precio)) return alert("Completa producto y precio");

  const venta = {
    producto, categoria, costo, precio, tipo, cliente, quincena,
    codigo: codigo || "",
    metodoPago,
    abonos: [], incobrable: false,
    fecha: new Date().toLocaleString("es-CO"),
    fechaISO: new Date().toISOString()
  };
  datos.ventas.push(venta);

  // Si es contado, sumar a caja
  if (tipo === "contado") {
    agregarMovimiento("ingreso", precio, `Venta: ${producto}`, categoria, null, "venta-contado");
  }

  guardar();
  document.getElementById("v-producto").value = "";
  document.getElementById("v-categoria").value = "";
  document.getElementById("v-costo").value = "";
  document.getElementById("v-precio").value = "";
  document.getElementById("v-cliente").value = "";
  document.getElementById("v-codigo").value = "";
  renderTodo();
}

function venderDelInventario(i) {
  const item = datos.inventario[i];
  if (item.cantidad <= 0) return alert("No hay stock");
  const tipo = confirm("¿Venta de contado?\n\nAceptar = Contado\nCancelar = Fiado");
  let cliente = "Contado", quincena = "";
  if (!tipo) {
    cliente = prompt("Nombre del cliente:");
    if (!cliente) return;
    const q = prompt("¿Quincena? Escribe 15 o 30:", "15");
    if (q !== "15" && q !== "30") return;
    quincena = q;
  }
  const metodoPago = tipo ? (prompt("Método de pago (efectivo/transferencia/nequi/daviplata):", "efectivo") || "efectivo") : "";
  datos.ventas.push({
    producto: item.producto, categoria: item.categoria,
    costo: item.costo, precio: item.precio,
    tipo: tipo ? "contado" : "quincena",
    cliente, quincena,
    metodoPago,
    codigo: item.codigo || "",
    abonos: [], incobrable: false,
    fecha: new Date().toLocaleString("es-CO"),
    fechaISO: new Date().toISOString()
  });
  if (tipo) {
    agregarMovimiento("ingreso", item.precio, `Venta: ${item.producto}`, item.categoria, null, "venta-contado");
  }
  item.cantidad -= 1;
  if (item.cantidad <= 0) datos.inventario.splice(i,1);
  guardar(); renderTodo();
  alert("✅ Venta registrada");
}

function procesarFotoAbono(ev) {
  const file = ev.target.files[0]; if (!file) return;
  const reader = new FileReader();
  reader.onload = (e) => {
    const img = new Image();
    img.onload = () => {
      const canvas = document.createElement("canvas");
      const maxW = 600;
      const scale = Math.min(1, maxW / img.width);
      canvas.width = img.width * scale;
      canvas.height = img.height * scale;
      canvas.getContext("2d").drawImage(img, 0, 0, canvas.width, canvas.height);
      fotoAbonoActual = canvas.toDataURL("image/jpeg", 0.7);
      document.getElementById("abono-foto-preview").innerHTML = `
        
        

          📸 Comprobante · Quitar
        
`;
    };
    img.src = e.target.result;
  };
  reader.readAsDataURL(file);
}
function quitarFotoAbono() {
  fotoAbonoActual = null;
  document.getElementById("abono-foto-preview").innerHTML = "";
  document.getElementById("abono-foto").value = "";
}
function confirmarAbono() {
  const { tipo, indice } = abonoActual;
  if (!tipo || indice === null) return;
  const item = tipo === "deuda" ? datos.deudas[indice] : datos.ventas[indice];
  if (!item) return;
  let monto = parseFloat(document.getElementById("abono-monto").value);
  const nota = document.getElementById("abono-nota").value.trim();
  const metodoPago = obtenerMedioPagoSeleccionado("medios-pago-abono");
  const saldoActual = calcularSaldo(item);

  if (isNaN(monto) || monto <= 0) return alert("Ingresa un monto válido");
  if (monto > saldoActual + 0.01) {
    if (!confirm(`El abono (${formato(monto)}) es mayor que el saldo (${formato(saldoActual)}). ¿Abonar solo ${formato(saldoActual)}?`)) return;
    monto = saldoActual;
  }
  if (!item.abonos) item.abonos = [];
  const nuevoAbono = {
    monto, nota, foto: fotoAbonoActual, metodoPago,
    fecha: new Date().toLocaleString("es-CO"),
    fechaISO: new Date().toISOString()
  };
  item.abonos.push(nuevoAbono);

  // Sumar a caja
  const nombre = tipo === "deuda" ? item.nombre : item.cliente;
  agregarMovimiento("ingreso", monto, `Abono de ${nombre}`, "Abonos", null, "abono");

  guardar(); renderTodo();
  cerrarModal("modal-abono");
  mostrarEstadoSync("ok", "Abono registrado");
  setTimeout(() => {
    if (confirm("¿Generar el recibo del abono?")) mostrarReciboAbono(tipo, indice, nuevoAbono);
  }, 400);
}

async function cargarHtml2Canvas() {
  if (window.html2canvas) return;
  return new Promise((resolve, reject) => {
    const script = document.createElement("script");
    script.src = "https://cdn.jsdelivr.net/npm/html2canvas@1.4.1/dist/html2canvas.min.js";
    script.onload = resolve;
    script.onerror = reject;
    document.head.appendChild(script);
  });
}
async function generarImagenDeElemento(elemento, nombreArchivo, descargarAuto) {
  try {
    mostrarEstadoSync("cargando", "Generando imagen...");
    await cargarHtml2Canvas();
    const canvas = await html2canvas(elemento, {
      backgroundColor: "#ffffff",
      scale: 2,
      useCORS: true,
      logging: false
    });
    const dataUrl = canvas.toDataURL("image/png");
    imagenGeneradaActual = { dataUrl, nombre: nombreArchivo };
    if (descargarAuto) {
      const link = document.createElement("a");
      link.href = dataUrl;
      link.download = nombreArchivo;
      link.click();
      mostrarEstadoSync("ok", "Imagen descargada");
    } else {
      document.getElementById("imagen-generada").src = dataUrl;
      document.getElementById("modal-imagen").classList.add("active");
      mostrarEstadoSync("ok", "Imagen lista");
    }
  } catch (e) {
    console.error(e);
    mostrarEstadoSync("error", "Error al generar imagen");
    alert("No se pudo generar la imagen. Verifica tu conexión a internet.");
  }
}

function agruparClientes() {
  const mapa = {};
  datos.ventas.forEach((v, idx) => {
    if (v.tipo !== "quincena") return;
    if (!mapa[v.cliente]) mapa[v.cliente] = { nombre: v.cliente, fiadoPendiente: 0, totalVendido: 0, compras: 0, ventasIdx: [] };
    mapa[v.cliente].fiadoPendiente += calcularSaldo(v);
    mapa[v.cliente].totalVendido += v.precio;
    mapa[v.cliente].compras += 1;
    mapa[v.cliente].ventasIdx.push(idx);
  });
  datos.deudas.forEach(d => {
    if (!mapa[d.nombre]) mapa[d.nombre] = { nombre: d.nombre, fiadoPendiente: 0, totalVendido: 0, compras: 0, ventasIdx: [], deudaAparte: 0 };
    mapa[d.nombre].deudaAparte = (mapa[d.nombre].deudaAparte || 0) + calcularSaldo(d);
  });
  return Object.values(mapa).sort((a,b) => (b.fiadoPendiente||0) - (a.fiadoPendiente||0));
}

function procesarFotoCliente(ev) {
  const file = ev.target.files[0]; if (!file) return;
  const reader = new FileReader();
  reader.onload = (e) => {
    const img = new Image();
    img.onload = () => {
      const canvas = document.createElement("canvas");
      const maxW = 300;
      const scale = Math.min(1, maxW / img.width);
      canvas.width = img.width * scale;
      canvas.height = img.height * scale;
      canvas.getContext("2d").drawImage(img, 0, 0, canvas.width, canvas.height);
      fotoClienteTemp = canvas.toDataURL("image/jpeg", 0.7);
      document.getElementById("fc-preview").innerHTML = ``;
    };
    img.src = e.target.result;
  };
  reader.readAsDataURL(file);
}

function proximaQuincena() {
  const hoy = new Date();
  const dia = hoy.getDate();
  let objetivo;
  if (dia < 15) objetivo = new Date(hoy.getFullYear(), hoy.getMonth(), 15);
  else if (dia < 30) objetivo = new Date(hoy.getFullYear(), hoy.getMonth(), 30);
  else objetivo = new Date(hoy.getFullYear(), hoy.getMonth()+1, 15);
  const dias = Math.ceil((objetivo - hoy) / (1000*60*60*24));

}

  const caja = calcularCaja();

    alertas.push({ icono: "💵", texto: `Tu caja está en ${formato(caja)}. ¿Necesitas revisar?`, tipo: "info", accion: "caja" });
  }

  if (alertas.length === 0) { el.innerHTML = ""; return; }

  el.innerHTML = `
    

      
🔔 Alertas inteligentes

      ${alertas.map(a => `
        

          
${a.icono}

          
${a.texto}

        

      `).join("")}
    
`;
}

function cerrarQuincena() {
  const q = document.getElementById("q-tipo").value;
  const sueldo = parseFloat(document.getElementById("q-sueldo").value) || 0;
  const descuentos = parseFloat(document.getElementById("q-descuentos").value) || 0;
  const neto = sueldo - descuentos;
  const ventasQ = datos.ventas.filter(v => v.tipo === "quincena" && v.quincena === q);
  const cobradoFiado = ventasQ.reduce((s,v) => s + calcularAbonado(v), 0);
  const gananciaFiado = ventasQ.reduce((s,v) => s + ((v.precio - v.costo) * (calcularAbonado(v) / (v.precio || 1))), 0);
  const saldoPendienteQ = ventasQ.reduce((s,v) => s + calcularSaldo(v), 0);
  const r = datos.responsabilidades;
  const totalResp = (r.recibos||0) + (r.comida||0) + (r.arriendo||0) + (r.otros||0);
  const restante = neto + cobradoFiado - totalResp;
  const hoy = new Date();
  const resumen = `📅 CIERRE QUINCENA DÍA ${q}
Fecha: ${hoy.toLocaleDateString("es-CO")} ${hoy.toLocaleTimeString("es-CO")}
━━━━━━━━━━━━━━━━━━━━━━━━━
Sueldo:              ${formato(sueldo)}
Descuentos:         -${formato(descuentos)}
─────────────────────────
Neto quincena:       ${formato(neto)}

Fiados cobrados:     ${formato(cobradoFiado)}
Saldo pendiente Q${q}:  ${formato(saldoPendienteQ)}

RESPONSABILIDADES:
  Recibos:          -${formato(r.recibos)}
  Comida:           -${formato(r.comida)}
  Arriendo:         -${formato(r.arriendo)}
  Otros:            -${formato(r.otros)}
  Total:            -${formato(totalResp)}
━━━━━━━━━━━━━━━━━━━━━━━━━
💰 RESTANTE:         ${formato(restante)}`;
  if (!confirm(resumen + "\n\n¿Cerrar esta quincena?")) return;
  datos.quincenas.push({
    fecha: new Date().toLocaleString("es-CO"),
    fechaISO: new Date().toISOString(),
    quincena: q, sueldo, descuentos, neto,
    cobradoFiado, gananciaFiado,
    responsabilidades: {...r}, totalResp, restante
  });
  datos.ventas = datos.ventas.filter(v => !(v.tipo === "quincena" && v.quincena === q && estaPagada(v)));
  guardar();
  document.getElementById("q-sueldo").value = "";
  document.getElementById("q-descuentos").value = "";
  renderTodo();
  document.getElementById("modal-contenido-texto").textContent = resumen;
  ultimoCierreQuincena = resumen;
  document.getElementById("modal-cierre").classList.add("active");
}

function descargarCSV(nombre, contenido) {
  const blob = new Blob(["\ufeff" + contenido], {type: "text/csv;charset=utf-8"});
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = nombre;
  a.click();
  URL.revokeObjectURL(url);
}
function exportarCSVVentas() {
  if (datos.ventas.length === 0) return alert("Sin ventas para exportar");
  const headers = ["Fecha", "Producto", "Categoría", "Cliente", "Tipo", "Quincena", "Costo", "Precio", "Ganancia", "Método pago", "Abonado", "Saldo"];
  const rows = datos.ventas.map(v => [
    v.fecha, v.producto, v.categoria || "", v.cliente, v.tipo, v.quincena || "",
    v.costo, v.precio, v.precio - v.costo, v.metodoPago || "efectivo",
    calcularAbonado(v), calcularSaldo(v)
  ]);
  const csv = [headers, ...rows].map(r => r.map(c => `"${String(c).replace(/"/g, '""')}"`).join(",")).join("\n");
  descargarCSV(`emma-ventas-${new Date().toISOString().slice(0,10)}.csv`, csv);
}

}
